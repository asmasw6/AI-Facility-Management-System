def determine_priority(category: str, sentiment: str, confidence: float, complaint_text: str) -> TicketPriority:
    """
    Determine the priority of a ticket based on its category.
    Adjust the logic as needed for your specific use case.
    """
    urgent_categories = {
        "Electricity Issue",
        "Security Complaint",
        "Plumbing Issue",
        "Water Supply Request",
    }

    urgent_keywords = [
        "fire",
        "smoke",
        "electrical hazard",
        "electric shock",
        "gas leak",
        "water leaking near electrical",
        "water leak near electrical",
        "flooding",
        "flood",
        "no electricity",
        "power outage",
        "elevator stuck",
        "emergency",
        "sparks",
        "short circuit",
    ]

    high_keywords = [
        "leaking",
        "water leak",
        "leakage",
        "completely stopped",
        "not working",
        "broken",
        "dangerous",
        "damage",
        "very hot",
        "no water",
    ]

    text = complaint_text.lower()

    # 1. Immediate safety/emergency issues
    if any(keyword in text for keyword in urgent_keywords):
        return TicketPriority.URGENT

    # 2. Serious facility categories
    if category in urgent_categories:
        return TicketPriority.HIGH

    # 3. Serious wording in the complaint
    if any(keyword in text for keyword in high_keywords):
        return TicketPriority.HIGH

    # 4. Customer sentiment
    if sentiment == "negative":
        return TicketPriority.MEDIUM

    # 5. Low model confidence
    if confidence < 0.5:
        return TicketPriority.MEDIUM

    return TicketPriority.LOW

    building: BuildingCreate,
    db: Session = Depends(get_db)
):

    existing_building = (
        db.query(Building)
        .filter(Building.building_name == building.building_name)
        .first()
    )

        raise HTTPException(
            status_code=400,
            detail="Building already exists"
        )

    new_building = Building(
        building_name=building.building_name
    )

    db.add(new_building)
    db.commit()
    db.refresh(new_building)

    return new_building

@app.post("/apartments/", response_model=ApartmentResponse)
def create_apartment(
    apartment: ApartmentCreate,
    db: Session = Depends(get_db)
):

    # Check building exists
    building = (
        db.query(Building)
        .filter(Building.building_id == apartment.building_id)
        .first()
    )

    if not building:
        raise HTTPException(
            status_code=404,
            detail="Building not found"
        )

    # Check apartment does not already exist
    existing_apartment = (
        db.query(Apartment)
        .filter(
            Apartment.building_id == apartment.building_id,
            Apartment.apartment_number == apartment.apartment_number
        )
        .first()
    )

    if existing_apartment:
        raise HTTPException(
            status_code=400,
            detail="Apartment already exists in this building"
        )

    new_apartment = Apartment(
        apartment_number=apartment.apartment_number,
        building_id=apartment.building_id
    )

    db.add(new_apartment)
    db.commit()
    db.refresh(new_apartment)

    return new_apartment

@app.post("/calls/{call_id}/end", response_model=CallResponse)
def end_call(call_id: int, call_end: CallEnd, db: Session = Depends(get_db)):
    call = db.query(Call).filter(Call.call_id == call_id).first()
    if not call:
        raise HTTPException(status_code=404, detail="Call not found")
    
    call.ended_at = datetime.utcnow()
    if call_end.transcript:
        call.transcript = call_end.transcript
    if call_end.audio_url:
        call.audio_url = call_end.audio_url

    if call.started_at:
        started = call.started_at.replace(tzinfo=None) if call.started_at.tzinfo else call.started_at
        call.duration = int((call.ended_at - call.started_at).total_seconds())
    
    db.commit()
    db.refresh(call)
    return call

@app.post("/process-call", response_model=TicketResponse)
async def process_call(request: ProcessCallRequest, db: Session = Depends(get_db)):
    # 1. Check BERT classifier
    if classifier is None:
        raise HTTPException(status_code=503, detail="BERT classifier is not loaded yet")    
    
    # 2. Get Customer
    customer = db.query(Customer).filter(Customer.customer_id == request.customer_id).first()
    if not customer:
        raise HTTPException(status_code=404, detail="Customer not found")     
    
    # 3. Get Call if provided
    if request.call_id is not None:
        call = db.query(Call).filter(Call.call_id == request.call_id).first()
        if not call:
            raise HTTPException(status_code=404, detail="Call not found")          
    
    # Optional: make sure the call belongs to this customer
        if call.customer_id != request.customer_id:
            raise HTTPException(
                status_code=400,
                detail="Call does not belong to this customer"
            )

    # 4. Get Apartment
    apartment = customer.apartment

    if not apartment:
        raise HTTPException(
            status_code=400,
            detail="Customer is not assigned to an apartment"
        )

    # 5. Get Building
    building = apartment.building

    if not building:
        raise HTTPException(
            status_code=400,
            detail="Apartment is not assigned to a building"
        )

    building_number = building.building_name.value
    apartment_number = apartment.apartment_number.value

   
    # 7. BERT: classification and confidence
    try:
        category, confidence = classifier.predict(request.text)
    except Exception as e:
        logger.error(f"Error during BERT prediction: {e}")
        raise HTTPException(status_code=500, detail="Error during BERT prediction")
    
    # 🔴🔴
    
    # 8. LLM: sentiment analysis
    sentiment = await get_sentiment_from_llm(request.text)
    print(f">>>>>>>>>>>>>> LLM sentiment analysis result: {sentiment}")
    
    # 9. Determine Ticket Priority
    priority = determine_priority(category, sentiment, confidence, request.text)
    

    # 10. LLM: Build prompt and call API
    
    ai_response = await generate_llm_response(
        complaint_text=request.text,
        category=category,
        sentiment=sentiment,
        priority=priority.value,
        building_number=building_number,     
        apartment_number=apartment_number,
    )
    # 11. Create Ticket
    new_ticket = Ticket(
        customer_id=request.customer_id,
        call_id=request.call_id,
        apartment_id=apartment.apartment_id,
        issue_description=request.text,
        predicted_category=category,
        priority=priority,
        status=TicketStatus.OPEN,
        ai_response=ai_response,
    )
    
    # 12. Save Ticket to DB
    
    db.add(new_ticket)
    db.commit()
    db.refresh(new_ticket)
    
    logger.info(
        f"Ticket #{new_ticket.ticket_id} created | "
        f"Building={building_number} | "
        f"Apartment={apartment_number} | "
        f"Category={category} | "
        f"Priority={priority.value}"
    )    
    
    return new_ticket

def update_ticket(ticket_id: int, payload: TicketUpdate, db: Session = Depends(get_db)):
# rr-----
    ticket = db.query(Ticket).filter(Ticket.ticket_id == ticket_id).first()
    if not ticket:
        raise HTTPException(status_code=404, detail="ticket not found")

    if payload.status is not None:
        ticket.status = payload.status
        if payload.status == TicketStatus.RESOLVED:
            ticket.resolved_at = datetime.utcnow()

    if payload.priority is not None:
        ticket.priority = payload.priority

    db.commit()
    db.refresh(ticket)
    return ticket
